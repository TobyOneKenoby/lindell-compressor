cmake_minimum_required(VERSION 3.22)

# Universal binary (Apple Silicon + Intel) on macOS
set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "")
set(CMAKE_OSX_DEPLOYMENT_TARGET "10.13" CACHE STRING "")

project(SA670 VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# JUCE: use libs/JUCE if present, otherwise download it
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/libs/JUCE/CMakeLists.txt")
    add_subdirectory(libs/JUCE)
else()
    include(FetchContent)
    FetchContent_Declare(JUCE
        GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
        GIT_TAG 8.0.4
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()

if(APPLE)
    set(SA670_FORMATS VST3 AU Standalone)
else()
    set(SA670_FORMATS VST3 Standalone)
endif()

juce_add_plugin(SA670
    COMPANY_NAME "SA Audio"
    PLUGIN_MANUFACTURER_CODE Saau
    PLUGIN_CODE S670
    BUNDLE_ID com.saaudio.sa670
    PRODUCT_NAME "SA-670"
    FORMATS ${SA670_FORMATS}
    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    VST3_CATEGORIES Fx Dynamics EQ
    AU_MAIN_TYPE kAudioUnitType_Effect
    COPY_PLUGIN_AFTER_BUILD TRUE)

target_sources(SA670 PRIVATE
    Source/PluginProcessor.cpp
    Source/PluginEditor.cpp
    Source/UI/Sprites.cpp)

# Panel artwork + photographic control sprites (generated by tools/make_assets.py)
file(GLOB SA670_SPRITES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Resources/sprites/*.png")
juce_add_binary_data(SA670Assets SOURCES Resources/panel.jpg ${SA670_SPRITES})

target_include_directories(SA670 PRIVATE Source)

target_compile_definitions(SA670 PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_DISPLAY_SPLASH_SCREEN=0)

target_link_libraries(SA670
    PRIVATE
        SA670Assets
        juce::juce_audio_utils
        juce::juce_dsp
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags)

# Offline test (Linux/macOS): cmake -DSA670_BUILD_TESTS=ON
option(SA670_BUILD_TESTS "Build offline test" OFF)
if(SA670_BUILD_TESTS)
    juce_add_console_app(SA670Test PRODUCT_NAME "SA670Test")
    target_sources(SA670Test PRIVATE tests/PluginTest.cpp)
    target_compile_definitions(SA670Test PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JucePlugin_Name="SA-670")
    target_link_libraries(SA670Test PRIVATE SA670 juce::juce_audio_utils juce::juce_dsp)
endif()
